const express = require('express');
const cors = require('cors');
const { createProxyMiddleware } = require('http-proxy-middleware');

const app = express();
const PORT = process.env.PORT || 3000;
const TARGET = process.env.API_TARGET || 'http://51.75.118.170:20041';

app.use(cors());
app.use(express.json());

app.use('/api', createProxyMiddleware({
  target: TARGET,
  changeOrigin: true,
  secure: false,
  logLevel: 'debug',
  pathRewrite: {
    '^/api': '',
  },
}));

app.use(express.static(__dirname + '/..'));

app.get('/health', (_req, res) => {
  res.json({ ok: true, target: TARGET });
});

app.listen(PORT, () => {
  console.log(`PRINCE MOVIE proxy running on http://localhost:${PORT}`);
  console.log(`Target API: ${TARGET}`);
});

module.exports = app;
